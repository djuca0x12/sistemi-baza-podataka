*Mehanizacija:

CREATE SEQUENCE MEHANIZACIJA_SEQ
  START WITH 6
  INCREMENT BY 1
  NOCACHE;
  
CREATE OR REPLACE TRIGGER TRG_MEHANIZACIJA_ID
BEFORE INSERT ON MEHANIZACIJA
FOR EACH ROW
BEGIN
  IF :NEW.IDMEHANIZACIJA IS NULL THEN
    SELECT MEHANIZACIJA_SEQ.NEXTVAL INTO :NEW.IDMEHANIZACIJA FROM DUAL;
  END IF;
END;
